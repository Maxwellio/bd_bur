CREATE TABLE public.spr_edizm (
  "key"    numeric(6) NOT NULL PRIMARY KEY,
  znach    varchar(30) NOT NULL,
  ops      varchar(100),
  sprcode  numeric(8),
  ked      numeric(14,7),
  grupa    numeric(3),
  /* Keys */
  CONSTRAINT spr_edizm_pkey
    PRIMARY KEY ("key"),
  /* Foreign keys */
  CONSTRAINT fk_sprcode1
    FOREIGN KEY (sprcode)
    REFERENCES public.common_spr(id)
);

CREATE UNIQUE INDEX uniqkedgrupa
  ON public.spr_edizm
  (ked, grupa);

CREATE TRIGGER trg_spredizm
  BEFORE UPDATE OR INSERT OR DELETE
  ON public.spr_edizm
  FOR EACH ROW
  EXECUTE PROCEDURE public.ftrg_spredizm();

ALTER TABLE public.spr_edizm
  OWNER TO postgres;

COMMENT ON TABLE public.spr_edizm
  IS 'Единицы измерений, системный справочник';

COMMENT ON COLUMN public.spr_edizm.sprcode
  IS 'код значения';

COMMENT ON COLUMN public.spr_edizm.ked
  IS 'доля относительно базовой ед.измер СИ (такой же категории)';

COMMENT ON COLUMN public.spr_edizm.grupa
  IS 'группировка единиц измерения';
