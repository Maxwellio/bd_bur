CREATE TABLE burnar.defnar (
  "key"        integer NOT NULL PRIMARY KEY,
  ownernar     integer NOT NULL,
  createdate   timestamp without time zone NOT NULL DEFAULT CURRENT_TIMESTAMP,
  brigadir     integer,
  narauthor    integer NOT NULL,
  nartype      integer NOT NULL,
  shifr        varchar(20),
  nm           varchar(50),
  checkparams  numeric,
  /* Keys */
  CONSTRAINT pk_defnarkey
    PRIMARY KEY ("key"),
  /* Foreign keys */
  CONSTRAINT fk_aothor
    FOREIGN KEY (narauthor)
    REFERENCES burnar.users(users_id), 
  CONSTRAINT fk_defnar_brigadir
    FOREIGN KEY (brigadir)
    REFERENCES burnar.people(id), 
  CONSTRAINT fk_defnaratrib
    FOREIGN KEY (nartype)
    REFERENCES burnar.sprnartype("key"), 
  CONSTRAINT fk_owner
    FOREIGN KEY (ownernar)
    REFERENCES burnar.spr_workers("key")
);
