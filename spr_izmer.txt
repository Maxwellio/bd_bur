CREATE TABLE public.spr_izmer (
  "key"        numeric NOT NULL PRIMARY KEY,
  znach        varchar(50) NOT NULL,
  ops          varchar(100),
  tip          numeric(4),
  pack_suffix  numeric,
  /* Keys */
  CONSTRAINT pk_izmercode
    PRIMARY KEY ("key")
);

CREATE TRIGGER trg_sprizmer_del_bef
  BEFORE DELETE
  ON public.spr_izmer
  FOR EACH ROW
  EXECUTE PROCEDURE public.ftrg_sprizmer_del_bef();

CREATE TRIGGER trg_sprizmer_ins_before
  BEFORE INSERT
  ON public.spr_izmer
  FOR EACH ROW
  EXECUTE PROCEDURE public.ftrg_sprizmer_ins_before();

CREATE TRIGGER trg_sprizmer_upd_bef
  BEFORE UPDATE
  ON public.spr_izmer
  FOR EACH ROW
  EXECUTE PROCEDURE public.ftrg_sprizmer_upd_bef();

ALTER TABLE public.spr_izmer
  OWNER TO postgres;

COMMENT ON TABLE public.spr_izmer
  IS 'Измерители, системный справочник';

COMMENT ON COLUMN public.spr_izmer.znach
  IS 'название измерителя';

COMMENT ON COLUMN public.spr_izmer.ops
  IS 'описание';

COMMENT ON COLUMN public.spr_izmer.tip
  IS 'Внутрення типизация для автотранспорта';

COMMENT ON COLUMN public.spr_izmer.pack_suffix
  IS 'Суффикс для текущего имени пакета';
