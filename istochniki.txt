CREATE TABLE public.istochniki (
  id          numeric NOT NULL PRIMARY KEY,
  parent      numeric,
  nm          varchar(600) NOT NULL,
  ops         varchar(700),
  info        varchar(400),
  ord         numeric(4),
  pref        varchar(10),
  newident    varchar(9),
  lnkfile     varchar(50),
  bukva       varchar(60),
  prim        varchar(2000),
  arhiv       numeric(1) DEFAULT 0,
  arhiv_dt    date,
  ord_new     numeric,
  gorod       varchar(50),
  izdat_name  varchar(100),
  data_izdat  numeric(4),
  /* Keys */
  CONSTRAINT istochniki_pkey
    PRIMARY KEY (id)
);

CREATE INDEX istoch_parid
  ON public.istochniki
  (parent);

CREATE INDEX istochniki_index01
  ON public.istochniki
  (id, parent);

CREATE INDEX istochniki_index_pid
  ON public.istochniki
  (parent, id);

CREATE INDEX pk_istoch_id
  ON public.istochniki
  (id);

CREATE TRIGGER trg_ist_ins_before
  BEFORE INSERT
  ON public.istochniki
  FOR EACH ROW
  EXECUTE PROCEDURE public.ftrg_ist_ins_before();

CREATE TRIGGER trg_ist_upd_before
  BEFORE UPDATE
  ON public.istochniki
  FOR EACH ROW
  EXECUTE PROCEDURE public.ftrg_ist_upd_before();

ALTER TABLE public.istochniki
