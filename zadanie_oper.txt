--Table: burnar.zadanie_oper

CREATE TABLE burnar.zadanie_oper (
  "key"             integer NOT NULL PRIMARY KEY,
  parent            integer,
  oper              numeric(7),
  operlifeid        numeric(7),
  begoperdate       timestamp without time zone,
  prnum             numeric(5) NOT NULL,
  whowork           integer,
  worktype          integer,
  razdel            numeric(8),
  narkey            integer NOT NULL,
  colorsel          numeric NOT NULL DEFAULT 0,
  volume            numeric(5,2),
  locked            numeric(1),
  operlifetype      numeric(7),
  operlifetypeoper  numeric,
  tipbur            numeric,
  /* Keys */
  CONSTRAINT pk_keyzadanie_oper
    PRIMARY KEY ("key"),
  /* Foreign keys */
  CONSTRAINT fk_parentop
    FOREIGN KEY (parent)
    REFERENCES burnar.zadanie_oper("key")
    ON DELETE CASCADE, 
  CONSTRAINT fk_workerz2
    FOREIGN KEY (whowork)
    REFERENCES burnar.spr_workers("key"), 
  CONSTRAINT fk_worktype5
    FOREIGN KEY (worktype)
    REFERENCES burnar.spr_worktype(id)
);

CREATE UNIQUE INDEX ni_key_parent
  ON burnar.zadanie_oper
  ("key", parent);

CREATE INDEX ni_parent
  ON burnar.zadanie_oper
  (parent);

CREATE INDEX ni_zad_narkey
  ON burnar.zadanie_oper
  (narkey);

CREATE INDEX nk_zadanie
  ON burnar.zadanie_oper
  (narkey, "key", operlifetypeoper, oper);

CREATE TRIGGER trg_log_oper_zad
  AFTER UPDATE OR INSERT OR DELETE
  ON burnar.zadanie_oper
  FOR EACH ROW
  EXECUTE PROCEDURE burnar.ftrg_log_oper_zad();

CREATE TRIGGER trg_zadanie_oper
  BEFORE UPDATE OR DELETE
  ON burnar.zadanie_oper
  FOR EACH ROW
  EXECUTE PROCEDURE burnar.ftrg_zadanie_oper();

ALTER TABLE burnar.zadanie_oper
  OWNER TO postgres;
