CREATE TABLE public.spr_oper (
  "key"          numeric(7) NOT NULL PRIMARY KEY,
  nm             varchar(512) NOT NULL,
  izmerzavis     numeric(7),
  nakolichzavis  numeric(6,2),
  kodop          varchar(6),
  shortnm        varchar(50),
  /* Keys */
  CONSTRAINT spr_oper_pkey
    PRIMARY KEY ("key")
);

CREATE UNIQUE INDEX un_kodop
  ON public.spr_oper
  (kodop);

  BEFORE INSERT
  ON public.spr_oper

CREATE TRIGGER trg_sproper_del_bef

CREATE TRIGGER trg_sproper_upd_bef
  BEFORE UPDATE
  ON public.spr_oper
  FOR EACH ROW
  EXECUTE PROCEDURE public.ftrg_sproper_upd_bef();

ALTER TABLE public.spr_oper

  IS 'Перечень операций';
