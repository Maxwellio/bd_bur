CREATE TABLE public.alg_operlife (
  alg          numeric NOT NULL,
  id_operlife  numeric NOT NULL,
  /* Foreign keys */
  CONSTRAINT fk_alg
    FOREIGN KEY (alg)
    REFERENCES public.algs("key")
    ON DELETE CASCADE, 
  CONSTRAINT fk_id_operlife
    FOREIGN KEY (id_operlife)
    REFERENCES public.operlife(id)
    ON DELETE CASCADE
);

ALTER TABLE public.alg_operlife
  OWNER TO postgres;

  IS 'код алгоритма';

COMMENT ON COLUMN public.alg_operlife.id_operlife
  IS 'код интервала жизни операции';
