CREATE TABLE public.normist (
  norm_id   numeric(7) NOT NULL,
  norm_ist  numeric(8) NOT NULL,
  /* Foreign keys */
  CONSTRAINT fk_normist_normid
    FOREIGN KEY (norm_id)
    REFERENCES public.norms("key")
    ON DELETE CASCADE
);

CREATE UNIQUE INDEX uk_istnorm
  ON public.normist
  (norm_id, norm_ist);

CREATE TRIGGER trg_normist_del_before
  BEFORE DELETE
  ON public.normist
  FOR EACH ROW
  EXECUTE PROCEDURE public.ftrg_normist_del_before();

CREATE TRIGGER trg_normist_ins_before
  BEFORE INSERT
  ON public.normist
  FOR EACH ROW
  EXECUTE PROCEDURE public.ftrg_normist_ins_before();

CREATE TRIGGER trg_normist_upd_before
  BEFORE UPDATE
  ON public.normist
  FOR EACH ROW
  EXECUTE PROCEDURE public.ftrg_normist_upd_before();

ALTER TABLE public.normist
  OWNER TO postgres;

COMMENT ON COLUMN public.normist.norm_id
  IS 'код нормы';

COMMENT ON COLUMN public.normist.norm_ist
  IS 'код источника нормирования ЭНБДТ';
