--Table: burnar.narvip

--DROP TABLE burnar.narvip;

CREATE TABLE burnar.narvip (
  oper         numeric(7) NOT NULL,
  begoperdate  timestamp without time zone NOT NULL,
  prnum        numeric(3) NOT NULL,
  whowork      integer NOT NULL,
  "key"        integer NOT NULL,
  worktype     integer,
  razdel       numeric(8),
  narkey       integer NOT NULL,
  sel          numeric(1),
  /* Keys */
  CONSTRAINT pk_keynarvip
    PRIMARY KEY ("key"),
  /* Foreign keys */
  CONSTRAINT fk_narkey4
    FOREIGN KEY (narkey)
    REFERENCES burnar.defnarvip(narkey)
    ON DELETE CASCADE, 
  CONSTRAINT fk_typework
    FOREIGN KEY (worktype)
    REFERENCES burnar.spr_worktype(id), 
  CONSTRAINT fk_workerv
    FOREIGN KEY (whowork)
    REFERENCES burnar.spr_workers("key")
);

ALTER TABLE burnar.narvip
  OWNER TO postgres;

COMMENT ON TABLE burnar.narvip
  IS 'полное содержание наряда в части выполнения';

COMMENT ON COLUMN burnar.narvip.oper
  IS 'код операции';

COMMENT ON COLUMN burnar.narvip.begoperdate
  IS 'фактическая дата начала действия операции (по нормативу), будет вычисляться, двигаться';

COMMENT ON COLUMN burnar.narvip.prnum
  IS 'порядковый номер операции (для ресурса)';

COMMENT ON COLUMN burnar.narvip.whowork
  IS 'кто выполнял работу (какая бригада или сила)  = ресурс';

COMMENT ON COLUMN burnar.narvip."key"
  IS 'код строки';

COMMENT ON COLUMN burnar.narvip.worktype
  IS 'Тип работы';

COMMENT ON COLUMN burnar.narvip.razdel
  IS 'код раздела каталога';

COMMENT ON COLUMN burnar.narvip.narkey
  IS 'код наряда';

COMMENT ON COLUMN burnar.narvip.sel
  IS 'выделен ли блок, 1-да, null-нет';

GRANT SELECT, INSERT, UPDATE, DELETE, REFERENCES, TRIGGER, TRUNCATE
  ON burnar.narvip
TO postgres;

GRANT SELECT, INSERT, UPDATE, DELETE, REFERENCES, TRIGGER, TRUNCATE
  ON burnar.narvip
TO burnar_role;
