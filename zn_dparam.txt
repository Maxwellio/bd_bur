CREATE TABLE public.zn_dparam (
  "key"       numeric(10) NOT NULL PRIMARY KEY,
  dparam      numeric(10) NOT NULL,
  znach       varchar(300) NOT NULL,
  ops         varchar(200),
  orderv      numeric(3),
  shortznach  varchar(100),
  kodznpar    varchar(9),
  sprcode     numeric(8),
  val2        numeric(10,2),
  val1        numeric(10,2),
  /* Keys */
  CONSTRAINT zn_dparam_pkey
    PRIMARY KEY ("key"),
  /* Foreign keys */
  CONSTRAINT fk_comspr1
    FOREIGN KEY (sprcode)
    REFERENCES public.common_spr(id), 
  CONSTRAINT fk_spr_id
    FOREIGN KEY (dparam)
    REFERENCES public.user_param("key")
);

CREATE TRIGGER trg_zndparam_del_bef
  BEFORE DELETE
  ON public.zn_dparam
  FOR EACH ROW
  EXECUTE PROCEDURE public.ftrg_zndparam_del_bef();

CREATE TRIGGER trg_zndparam_upd_bef
  BEFORE UPDATE
  ON public.zn_dparam
  FOR EACH ROW
  EXECUTE PROCEDURE public.ftrg_zndparam_upd_bef();

CREATE TRIGGER trg_znnormoobraz_par_ins_bef
  BEFORE INSERT
  ON public.zn_dparam
  FOR EACH ROW
  EXECUTE PROCEDURE public.ftrg_znnormoobraz_par_ins_bef();

ALTER TABLE public.zn_dparam
  OWNER TO postgres;
