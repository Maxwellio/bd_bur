  norm_id  numeric NOT NULL,
  stroka   varchar(1024) NOT NULL,
  /* Foreign keys */
  CONSTRAINT fk_normkey1
    FOREIGN KEY (norm_id)
    REFERENCES public.norms("key")
    ON DELETE CASCADE
);

  IS 'Хранимая строка';
