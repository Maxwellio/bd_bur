CREATE TABLE burnar.znrepparsv (
  defnar      integer NOT NULL,
  znval       numeric(8),
  val         numeric(9,3),
  valstr      varchar(150),
  repid       integer NOT NULL,
  parcode     numeric(8) NOT NULL,
  prnum       numeric(3) NOT NULL,
  repparname  varchar(20),
  period_key  integer,
  /* Keys */
  CONSTRAINT uk_znrepparsv
    UNIQUE (defnar, repid, parcode, prnum, period_key),
  /* Foreign keys */
  CONSTRAINT fk_defnar6
    FOREIGN KEY (defnar)
    REFERENCES burnar.defnar("key")
    ON DELETE CASCADE, 
  CONSTRAINT fk_period_key
    FOREIGN KEY (period_key)
    REFERENCES burnar.vipolnenie_period("key")
    ON DELETE CASCADE, 
  CONSTRAINT fk_repid6
    FOREIGN KEY (repid)
    REFERENCES burnar.reports(id)
    ON DELETE CASCADE
);

CREATE INDEX nk_znrepparsv1
  ON burnar.znrepparsv
  (period_key, repid, parcode);

CREATE TRIGGER trg_znrepparsv_up_after
  AFTER UPDATE
  ON burnar.znrepparsv
  FOR EACH ROW
  EXECUTE PROCEDURE burnar.ftrg_znrepparsv_up_after();

ALTER TABLE burnar.znrepparsv
