--Procedure: burnar.change_password_strict(varchar, varchar)

--DROP PROCEDURE burnar.change_password_strict(varchar, varchar);

CREATE OR REPLACE PROCEDURE burnar.change_password_strict
(
  IN  p_username      varchar,
  IN  p_new_password  varchar
)
AS $$
DECLARE
BEGIN

  --смена пароля пользователя для админов. не зная пароль пользователя
  update burnar.users
  set password = burnar.get_hash(p_username, p_new_password)
  where UPPER(ora_name) = UPPER(p_username); 
        --and password = burnar.get_hash(p_username, p_old_password);
  
exception
  when NO_DATA_FOUND then
  RAISE NOTICE '-20000,Invalid username/password.';


END
$$
LANGUAGE 'plpgsql';

ALTER PROCEDURE burnar.change_password_strict(varchar, varchar)
  OWNER TO postgres;
