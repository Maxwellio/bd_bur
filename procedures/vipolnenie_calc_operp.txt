--Procedure: burnar.vipolnenie_calc_operp(bigint, integer, numeric, numeric, numeric, varchar, numeric, numeric)

--DROP PROCEDURE burnar.vipolnenie_calc_operp(bigint, integer, numeric, numeric, numeric, varchar, numeric, numeric);

CREATE OR REPLACE PROCEDURE burnar.vipolnenie_calc_operp
(
  IN     akey         bigint,
  IN     aoperlifeid  integer,
  INOUT  aizmer       numeric,
  INOUT  normaed      numeric,
  INOUT  normaob      numeric,
  INOUT  istochniki   varchar,
  INOUT  aot          numeric,
  INOUT  ado          numeric
)
AS $$
declare 
  Pars public.TContPar[];
  tmp  public.TNormaObj; --значение вектора нормы
  i    numeric := 0;
  vol  numeric; --объём работ 
  rec1 record;
  rec2 record;
  rec record;  
  mass_ell public.tnorma;
  mass_id numeric;
begin

 --обнуляем переменные
  istochniki := '';
  normaed := 0;
  normaob := 0;
  aot := null;
  ado := null;
  
  --получаем сохраненные параметры листовой операции
  select array_agg((t.parcode, t.znach)::public.TCONTPAR) into pars
  from burnar.vipolnenie_param t
  where t.vip_key = aKey;
   
  ---удаляем старые нормы
  update burnar.vipolnenie_norm 
  set NORMA = null 
  where vip_key = akey;
  
  -- получаем результирующий объект
  tmp := public.clcleafobj(aoperlifeid, aIzmer, Pars)::public.TNormaObj;  



  -- сохраняем новый вектор нормы
--  mass_id := 1; -- это херня,нужно брать номер элемента в массиве
-- берем реальный номер элемента в массиве, для нас это будет PRNUM
  for rec in (select generate_subscripts(tmp.aNorma,1) as idx, (unnest(tmp.aNorma)).norma as val)
  loop
--        raise notice 'idx = %', rec.idx;  
--        raise notice 'val = %', rec.val;
        
     update burnar.vipolnenie_norm
         set NORMA = rec.val     
     where vip_key = akey
           and PRNUM = rec.idx;
  end loop;

/*--это не правильно 
  foreach mass_ell in array tmp.anorma   --for i in 1 .. tmp.anorma.Count loop update burnar.vipolnenie_norm set NORMA = tmp.aNorma(i).norma where vip_key = akey and PRNUM = i; end loop; 
  loop

     update burnar.vipolnenie_norm
         set NORMA = mass_ell.norma --tmp.aNorma(i).norma    
     where vip_key = akey
           and PRNUM = mass_id; --i;
     
     raise notice 'norma = %',mass_ell.norma; 
     raise notice 'prnum = %',mass_id;     
     
     mass_id := mass_id + 1;  
  end loop;
*/
  
  -- теперь нужно найти уникальные источники и сохранить их
  --но сначала удалим старые
  delete from burnar.vipolnenie_ist 
  where vip_key = akey;
  
  -- а вот и сохраняем новые
  for rec1 in ( select distinct (unnest(tmp.aIst)).ist as ist )
  loop
    if rec1.ist is not null then
      insert into burnar.vipolnenie_ist (vip_key, ist)
      values (akey, rec1.ist);
    end if;   
  end loop;      

  --сохраняем строки
  --прежде удалим старые
  delete from burnar.vipolnenie_stroka 
  where vip_KEY = akey;
  
  for rec2 in (select stroka, row_number() OVER () as RM
               from unnest(tmp.aSTROKA)
              ) 
  loop
    if rec2.stroka is not null then
      insert into burnar.vipolnenie_stroka (vip_key, stroka, prnum) 
      values (akey, rec2.stroka, rec2.rm);
    end if;  
  end loop;

    /*Тянем значения параметров для данной операции*/
begin
  select  t.znach
    into strict aot
    from burnar.vipolnenie_param t
   where t.vip_key = aKey and t.parcode in (select * from burnar.ot_params);
exception
when no_data_found then aot := null; 
end;

begin
  select  t.znach
    into strict aDo
    from burnar.vipolnenie_param t
   where t.vip_key = aKey and t.parcode in (select * from burnar.do_params)   ;
exception
when no_data_found then aDo := null; 
end;
  

  istochniki := burnar.vipolnenie_getoperist(aKey);  --ltrim(istochniki,';');
  normaed    := round(tmp.aNorma[1].norma/60,2);
  normaob    := round(tmp.aNorma[2].norma/60,2);

  
  exception when others then 
  begin
    RAISE EXCEPTION 'Код строки= % %', aKey, SQLERRM;
  end;  
  raise;

end ;
$$
LANGUAGE 'plpgsql';

ALTER PROCEDURE burnar.vipolnenie_calc_operp(bigint, integer, numeric, numeric, numeric, varchar, numeric, numeric)
  OWNER TO postgres;
