CREATE TABLE public.comboperparam (
  operlifeid  numeric(7) NOT NULL,
  param       numeric(10) NOT NULL,
  prnum       numeric(2) NOT NULL,
  /* Keys */
  CONSTRAINT comboperparam_pkey
    PRIMARY KEY (operlifeid, param),
  /* Foreign keys */
  CONSTRAINT fk_comboperparam_operlifeid
    FOREIGN KEY (operlifeid)
    REFERENCES public.operlife(id), 
  CONSTRAINT fk_comboperparam_userparam
    FOREIGN KEY (param)
    REFERENCES public.user_param("key")
);

CREATE UNIQUE INDEX pk1
  ON public.comboperparam
  (operlifeid, param);

CREATE TRIGGER trg_comboperparam_del_bef
  BEFORE DELETE
  ON public.comboperparam
  FOR EACH ROW
  EXECUTE PROCEDURE public.ftrg_comboperparam_del_bef();

CREATE TRIGGER trg_comboperparam_ins_bef
  BEFORE INSERT
  ON public.comboperparam
  FOR EACH ROW
  EXECUTE PROCEDURE public.ftrg_alg_calc2_ins_before();

CREATE TRIGGER trg_comboperparam_upd_bef
  BEFORE UPDATE
  ON public.comboperparam
  FOR EACH ROW
  EXECUTE PROCEDURE public.ftrg_comboperparam_upd_bef();

ALTER TABLE public.comboperparam
  OWNER TO postgres;

COMMENT ON TABLE public.comboperparam
  IS 'Настроечная таблица. Нормообразующие параметры (только для комбинационных)';

COMMENT ON COLUMN public.comboperparam.operlifeid
  IS 'код интервала жизни операции';

COMMENT ON COLUMN public.comboperparam.param
  IS 'код нормообразующего параметра';

COMMENT ON COLUMN public.comboperparam.prnum
  IS 'порядковый номер параметра';
