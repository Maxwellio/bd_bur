CREATE TABLE burnar.vipolnenie_oper (
  "key"             bigint NOT NULL PRIMARY KEY,
  parent            bigint,
  oper              numeric(7),
  operlifeid        numeric(7),
  begoperdate       timestamp without time zone,
  prnum             numeric(5) NOT NULL,
  whowork           integer,
  worktype          integer,
  razdel            numeric(8),
  narkey            integer NOT NULL,
  colorsel          numeric NOT NULL DEFAULT 0,
  volume            numeric(5,2),
  locked            numeric(1),
  operlifetype      numeric(7),
  period            integer,
  operlifetypeoper  numeric,
  editdate          numeric(1),
  priznak           numeric(1) NOT NULL DEFAULT 0,
  zadoperkey        numeric(7),
  tipbur            numeric,
  /* Keys */
  CONSTRAINT pk_keyvipolnenie_oper
    PRIMARY KEY ("key"),
  /* Foreign keys */
  CONSTRAINT fk_narkey8
    FOREIGN KEY (narkey)
    REFERENCES burnar.defnarvip(narkey)
    ON DELETE CASCADE, 
  CONSTRAINT fk_parentop2
    FOREIGN KEY (parent)
    REFERENCES burnar.vipolnenie_oper("key")
    ON DELETE CASCADE, 
  CONSTRAINT fk_period
    FOREIGN KEY (period)
    REFERENCES burnar.vipolnenie_period("key")
    ON DELETE SET NULL, 
  CONSTRAINT fk_workerz4
    FOREIGN KEY (whowork)
    REFERENCES burnar.spr_workers("key"), 
  CONSTRAINT fk_worktype8
    FOREIGN KEY (worktype)
    REFERENCES burnar.spr_worktype(id)
);
